output application/json

fun yamlToJson(obj, prefix="") =
  (obj pluck ((value, key) ->
    if (value is Object)
      yamlToJson(value, prefix ++ (if (prefix == "") "" else ".") ++ (key as String))
    else
      { ((prefix ++ (if (prefix == "") "" else ".") ++ (key as String))): value }
  )) reduce ((item, acc = {}) -> acc ++ item)
